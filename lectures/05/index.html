jQuery · DOM 직접 수정

let count = 0;
const button = $('<button>').appendTo(host);
button.text(`Clicked ${count} times`);
button.on('click', () => {
  count += 1;
  button.text(`Clicked ${count} times`);
});

count 증가 → button.text(...) 호출

React · 상태에 따른 UI

function Counter() {
  const [count, setCount] = useState(0);
  return (
    <button onClick={() => setCount(count + 1)}>
      Clicked {count} times
    </button>
  );
}

  • 상태 변경 함수는 현재 실행의 값을 바꾸지 않고 다음 렌더를 요청
  • 기존 핸들러와 타이머는 만들어질 때 받은 이전 값을 계속 참조
  • 이벤트 처리 도중 값이 바뀌지 않아 같은 기준으로 계산 가능

  • React는 각 함수에 앞선 업데이트까지 반영한 상태 값을 전달

상태의 불변성

기존 값을 직접 바꾸지 않고 새 값으로 상태를 교체하는 원칙

상태 불변성 - 기존 상태를 직접 수정하지 않고 새 값으로 교체하는 원칙
  • 배열·객체 상태는 변경 내용을 담은 새 배열·객체로 갱신
  • 직접 수정하면 React가 변경을 놓치거나 이전 렌더의 값이 바뀔 수 있음
  • 이전 상태를 보존해 지난 렌더에서 사용한 값의 변경 방지
  • 새 참조를 전달해 React가 상태 변경을 인식할 수 있도록 함
  • 변경하지 않은 항목은 기존 객체를 재사용해 불필요한 복사 감소

Effect 정리 함수

이전 Effect에서 시작한 외부 연결을 해제하는 함수

정리 함수 - Effect에서 시작한 작업을 종료하는 함수
  • Effect 안에서 함수를 반환하면 React가 필요할 때 실행
  • 의존성 배열의 값이 바뀌면, 이전 작업을 종료한 뒤 새 값으로 시작

  • 타이머 중복 실행과 불필요하게 남아 있는 연결 방지

import { useEffect, useState } from 'react';

function Poller({ pName }) {
  useEffect(() => {
    const timer = setInterval(() => console.log('poll', pName), 1000);
    return () => {
      clearInterval(timer);
      console.log('stop', pName);
    };
  }, [pName]);
  return <p>조회 대상: {pName}</p>;
}
export default function App() {
  const [pName, setPName] = useState('사과');
  return (
    <>
      <button onClick={() => setPName('배')}>값 변경</button>
      <button onClick={() => setPName(null)}>컴포넌트 제거</button>
      {pName && <Poller pName={pName} />}
    </>
  );
}

커스텀 Hook

반복되는 Hook 로직을 함수로 추출해 재사용하는 방법

커스텀 Hook - Hook을 사용하는 로직을 재사용하는 함수
  • 이름을 use로 시작하고, 함수 안에서 useState 등 기존 Hook 사용
  • 여러 컴포넌트에서 같은 로직을 가져다 써 중복 코드 감소
  • 공통 로직을 한곳에서 수정해 유지보수 단순화
  • 같은 Hook을 호출해도 상태는 각 컴포넌트에서 따로 유지

React · App.jsx

import { useState } from 'react';

function useQuantity() {
  const [value, setValue] = useState(1);
  return [value, () => setValue((v) => v + 1)];
}
function Quantity({ pName }) {
  const [value, increment] = useQuantity();
  return <button onClick={increment}>{`${pName}: ${value}`}</button>;
}
export default function App() {
  return (
    <>
      <Quantity pName="사과" />
      <Quantity pName="배" />
    </>
  );
}
